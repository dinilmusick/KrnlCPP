cmake_minimum_required(VERSION 3.16)
project(KrnlCpp VERSION 0.2.0 LANGUAGES CXX)
add_library(krnlcpp INTERFACE)
target_include_directories(krnlcpp INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(krnlcpp INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 foreach(test IN ITEMS kernel_test concurrency_test lifecycle_test lifecycle_matrix_test lifecycle_paths_test lifecycle_rollback_test)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test}.cpp")
   add_executable(krnlcpp_${test} tests/${test}.cpp)
   target_link_libraries(krnlcpp_${test} PRIVATE krnlcpp Threads::Threads)
   add_test(NAME krnlcpp_${test} COMMAND krnlcpp_${test})
  endif()
 endforeach()
endif()
install(DIRECTORY include/ DESTINATION include)
install(TARGETS krnlcpp EXPORT KrnlCppTargets)
install(EXPORT KrnlCppTargets NAMESPACE krnl:: DESTINATION lib/cmake/KrnlCpp)
include(CMakePackageConfigHelpers)
configure_package_config_file(${CMAKE_CURRENT_SOURCE_DIR}/cmake/KrnlCppConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/KrnlCppConfig.cmake INSTALL_DESTINATION lib/cmake/KrnlCpp)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/KrnlCppConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/KrnlCppConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/KrnlCppConfigVersion.cmake DESTINATION lib/cmake/KrnlCpp)
