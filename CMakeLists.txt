cmake_minimum_required(VERSION 3.16)
project(KrnlCpp LANGUAGES CXX)
add_library(krnlcpp INTERFACE)
target_include_directories(krnlcpp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(krnlcpp INTERFACE cxx_std_17)
include(CTest)

  target_link_libraries(krnlcpp_lifecycle_test PRIVATE krnlcpp)
  add_test(NAME krnlcpp_lifecycle COMMAND krnlcpp_lifecycle_test)

  add_executable(krnlcpp_lifecycle_matrix_test tests/lifecycle_matrix_test.cpp)
  target_link_libraries(krnlcpp_lifecycle_matrix_test PRIVATE krnlcpp)
  add_test(NAME krnlcpp_lifecycle_matrix COMMAND krnlcpp_lifecycle_matrix_test)
  add_executable(krnlcpp_lifecycle_paths_test tests/lifecycle_paths_test.cpp)
  target_link_libraries(krnlcpp_lifecycle_paths_test PRIVATE krnlcpp)
  add_test(NAME krnlcpp_lifecycle_paths COMMAND krnlcpp_lifecycle_paths_test)
endif()
